5. 增加注册表状态读取过程
在 MainForm.pas 中增加：
procedure TfrmMain.SyncAutoStart;
const
  RunKey = 'Software\Microsoft\Windows\CurrentVersion\Run';
  RunName = 'WinLinuxTool_AutoStart';
var
  Reg: TRegistry;
  ExeValue: string;
begin
  miAutoStart.Checked := False;

  Reg := TRegistry.Create;
  try
    Reg.RootKey := HKEY_CURRENT_USER;

    if not Reg.OpenKey(RunKey, False) then Exit;

    if Reg.ValueExists(RunName) then
    begin
      ExeValue := '"' + ExpandFileName(Application.ExeName) + '"';

      { EXE 移动后自动更新启动路径 }
      if not SameText(Reg.ReadString(RunName), ExeValue) then
        Reg.WriteString(RunName, ExeValue);

      miAutoStart.Checked := True;
    end;

    Reg.CloseKey;
  finally
    Reg.Free;
  end;
end;


6. 增加菜单点击事件
procedure TfrmMain.miAutoStartClick(Sender: TObject);
const
  RunKey = 'Software\Microsoft\Windows\CurrentVersion\Run';
  RunName = 'WinLinuxTool_AutoStart';
var
  Reg: TRegistry;
  ExeValue: string;
begin
  Reg := TRegistry.Create;
  try
    try
      Reg.RootKey := HKEY_CURRENT_USER;

      if not Reg.OpenKey(RunKey, True) then
      begin
        ShowMessage('无法打开 Windows 开机启动注册表项！');
        Exit;
      end;

      if Reg.ValueExists(RunName) then
      begin
        { 取消开机启动 }
        Reg.DeleteValue(RunName);
        miAutoStart.Checked := False;
      end
      else
      begin
        { 开启开机启动 }
        ExeValue := '"' + ExpandFileName(Application.ExeName) + '"';

        Reg.WriteString(RunName, ExeValue);
        miAutoStart.Checked := True;
      end;

      Reg.CloseKey;
    except
      on E: Exception do
        ShowMessage('设置开机启动失败：' + E.Message);
    end;
  finally
    Reg.Free;
  end;
end;
